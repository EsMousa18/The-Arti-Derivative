    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const tabs = document.querySelectorAll('.tab-button');
            const contentArea = document.getElementById('content-area');

            /**
             * Handles the tab switching logic.
             * @param {string} targetTabId - The ID of the content section to show.
             */
            function switchTab(targetTabId) {
                // 1. Deactivate all tab buttons and hide all content sections
                tabs.forEach(button => {
                    button.classList.remove('active');
                    // Ensure inactive buttons are the pink color
                    button.classList.add('text-arti-pink');
                    button.classList.remove('text-white');
                });
                document.querySelectorAll('.tab-content').forEach(content => {
                    content.classList.add('hidden');
                });

                // 2. Activate the selected button
                const activeTabButton = document.querySelector(`.tab-button[data-tab="${targetTabId}"]`);
                if (activeTabButton) {
                    activeTabButton.classList.add('active');
                    // Ensure active button is white text
                    activeTabButton.classList.remove('text-arti-pink');
                    activeTabButton.classList.add('text-white');
                }

                // 3. Show the target content section
                const activeContent = document.getElementById(targetTabId);
                if (activeContent) {
                    activeContent.classList.remove('hidden');
                }
            }

            // Set up click listeners for all tab buttons
            tabs.forEach(button => {
                button.addEventListener('click', () => {
                    const targetId = button.getAttribute('data-tab');
                    switchTab(targetId);
                });
            });
            
            // Set the initial active tab on page load
            switchTab('mission');
        });
    </script>
